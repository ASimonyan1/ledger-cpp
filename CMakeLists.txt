cmake_minimum_required(VERSION 3.20)
project(ledger_cpp VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ENABLE_SANITIZERS "Enable ASan and UBSan" OFF)
include(cmake/Dependencies.cmake)
find_package(PostgreSQL REQUIRED)
find_package(Threads REQUIRED)
add_library(ledger src/ledger.cpp)
target_include_directories(ledger PUBLIC include)
target_link_libraries(ledger PUBLIC PostgreSQL::PostgreSQL)
add_executable(ledger_server src/main.cpp)
target_link_libraries(ledger_server PRIVATE ledger web_dependencies Threads::Threads)
if(MSVC)
  target_compile_definitions(ledger_server PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
if(WIN32)
  target_link_libraries(ledger_server PRIVATE ws2_32 crypt32)
endif()
foreach(target ledger ledger_server)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    if(ENABLE_SANITIZERS)
      target_compile_options(${target} PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${target} PUBLIC -fsanitize=address,undefined)
    endif()
  endif()
endforeach()
enable_testing()
find_package(Python3 COMPONENTS Interpreter)
if(Python3_FOUND)
  add_test(NAME http_and_database_contract COMMAND ${Python3_EXECUTABLE}
    ${CMAKE_SOURCE_DIR}/tests/integration.py --server $<TARGET_FILE:ledger_server>)
  set_tests_properties(http_and_database_contract PROPERTIES TIMEOUT 120)
endif()
